12.1 플립플롭과 시프트 레지스터 설계

12.1.1 8비트 Register

8개의 D Flip-Flop이 병렬로 구성된 8비트 register이다. posedge clk에서 din[7:0] 전체가 동시에 저장되고, 다음 clock edge까지 dout[7:0]은 유지된다.

din[7:0]→
8 × DFF
posedge clk
→dout[7:0]
합성 구조
8비트 Register = DFF 8개. Reset MUX가 각 DFF 입력 또는 전용 reset 구조로 매핑될 수 있다.
코드 — 8비트 레지스터
Reset은 synchronous active-high 방식으로 통일한다.
module reg_8b (
    input        clk,
    input        rst,
    input  [7:0] din,
    output reg [7:0] dout
);

always @(posedge clk) begin
    if (rst)
        dout <= 8'h00;
    else
        dout <= din;
end

endmodule
코드 — 테스트벤치
동기식 reset은 clock edge에서 반영되므로 테스트벤치도 posedge를 기준으로 reset을 검증한다.
`timescale 1ns/1ps

module tb_reg_8b;

reg        clk;
reg        rst;
reg  [7:0] din;
wire [7:0] dout;
integer i;

reg_8b dut (
    .clk (clk),
    .rst (rst),
    .din (din),
    .dout(dout)
);

initial begin
    clk = 1'b0;
    forever #5 clk = ~clk;   // 10 ns period
end

initial begin
    rst = 1'b1;
    din = 8'h00;

    // 동기식 reset이므로 posedge clk에서 reset이 반영된다.
    repeat (2) @(posedge clk);
    rst = 1'b0;

    for (i = 0; i < 16; i = i + 1) begin
        @(negedge clk);
        din = i;
    end

    repeat (2) @(posedge clk);
    $finish;
end

endmodule

12.1.2 SIPO 8비트 Shift Register

직렬 입력 sdin이 LSB로 들어오고 기존 비트가 매 clock마다 MSB 방향으로 한 칸 이동한다. 중요한 점은 모든 bit가 같은 edge 직전의 값을 기준으로 이동해야 하므로 nonblocking 할당을 사용해야 한다.

sdin→q[0]→q[1]→ … →q[7]
코드 — 시프트 연산자 방식
module shift_reg8b_shop (
    input        clk,
    input        rst,
    input        sdin,
    output [7:0] dout
);

reg [7:0] q;
assign dout = q;

always @(posedge clk) begin
    if (rst)
        q <= 8'h00;
    else
        q <= (q << 1) | {7'b0, sdin};
end

endmodule
코드 — 결합 연산자 방식
{q[6:0], sdin}은 q[7]←q[6], …, q[1]←q[0], q[0]←sdin과 동일하다.
module shift_reg8b_conop (
    input        clk,
    input        rst,
    input        sdin,
    output [7:0] dout
);

reg [7:0] q;
assign dout = q;

always @(posedge clk) begin
    if (rst)
        q <= 8'h00;
    else
        q <= {q[6:0], sdin};
end

endmodule
코드 — for 반복문 방식
for는 시간적으로 7번 반복하는 것이 아니라 합성 시 병렬 연결되는 register 이동 구조를 기술한다.
module shift_reg8b_for (
    input        clk,
    input        rst,
    input        sdin,
    output [7:0] dout
);

reg [7:0] q;
integer i;

assign dout = q;

always @(posedge clk) begin
    if (rst) begin
        q <= 8'h00;
    end
    else begin
        // 각 단계가 이전 단계 값을 저장: q[7]←q[6], q[6]←q[5], q[5]←q[4], q[4]←q[3], q[3]←q[2], q[2]←q[1], q[1]←q[0]
        for (i = 7; i > 0; i = i - 1)
            q[i] <= q[i-1];

        q[0] <= sdin;
    end
end

endmodule
코드 — Shift Register 테스트벤치
`timescale 1ns/1ps

module tb_shift_reg8b;

reg clk;
reg rst;
reg sdin;
wire [7:0] dout;

shift_reg8b_conop dut (
    .clk (clk),
    .rst (rst),
    .sdin(sdin),
    .dout(dout)
);

initial begin
    clk = 1'b0;
    forever #5 clk = ~clk;
end

initial begin
    rst  = 1'b1;
    sdin = 1'b0;
    repeat (2) @(posedge clk);
    rst = 1'b0;

    // 1,0,1,1,0,0,1,0 입력
    @(negedge clk) sdin = 1'b1;
    @(negedge clk) sdin = 1'b0;
    @(negedge clk) sdin = 1'b1;
    @(negedge clk) sdin = 1'b1;
    @(negedge clk) sdin = 1'b0;
    @(negedge clk) sdin = 1'b0;
    @(negedge clk) sdin = 1'b1;
    @(negedge clk) sdin = 1'b0;

    repeat (3) @(posedge clk);
    $finish;
end

endmodule

12.1.3 8비트 Ring Counter

Ring Counter는 마지막 bit를 다시 처음 bit로 귀환시켜 하나의 ‘1’이 순환하도록 만든다. 초기값은 0000_0001처럼 반드시 one-hot 상태를 줘야 한다.

0000_0001→0000_0010→0000_0100→ … →1000_0000→0000_0001
코드 — 결합 연산자
module ring_cnt8b_conop (
    input        clk,
    input        rst,
    output reg [7:0] cnt
);

always @(posedge clk) begin
    if (rst)
        cnt <= 8'b0000_0001;
    else if ((cnt == 8'h00) || (cnt == 8'h80))
        cnt <= 8'b0000_0001;
    else
        cnt <= {cnt[6:0], cnt[7]};
end

endmodule
코드 — 시프트 연산자
module ring_cnt8b_shop (
    input        clk,
    input        rst,
    output reg [7:0] cnt
);

always @(posedge clk) begin
    if (rst)
        cnt <= 8'b0000_0001;
    else if ((cnt == 8'h00) || (cnt == 8'h80))
        cnt <= 8'b0000_0001;
    else
        cnt <= (cnt << 1) | {7'b0, cnt[7]};
end

endmodule
코드 — for 반복문
module ring_cnt8b_for (
    input        clk,
    input        rst,
    output reg [7:0] cnt
);

integer i;

always @(posedge clk) begin
    if (rst) begin
        cnt <= 8'b0000_0001;
    end
    else if ((cnt == 8'h00) || (cnt == 8'h80)) begin
        cnt <= 8'b0000_0001;
    end
    else begin
        for (i = 7; i > 0; i = i - 1)
            cnt[i] <= cnt[i-1];

        cnt[0] <= cnt[7];
    end
end

endmodule
코드 — 테스트벤치
`timescale 1ns/1ps

module tb_ring_cnt8b;

reg clk;
reg rst;
wire [7:0] cnt;

ring_cnt8b_conop dut (
    .clk(clk),
    .rst(rst),
    .cnt(cnt)
);

initial begin
    clk = 1'b0;
    forever #10 clk = ~clk;
end

initial begin
    rst = 1'b1;
    repeat (2) @(posedge clk);
    rst = 1'b0;

    repeat (12) @(posedge clk);

    rst = 1'b1;
    @(posedge clk);
    rst = 1'b0;

    repeat (4) @(posedge clk);
    $finish;
end

endmodule

12.1.4 좌우 Ring Counter

mode=1이면 왼쪽 회전, mode=0이면 오른쪽 회전한다. 방향이 바뀔 때 one-hot 상태가 경계값에 걸리는 경우 설정 조건에 맞춰 각각 1 또는 128로 복귀시킨다.

코드 — Shift operator 방식
module LRS_ring_cnt8b_shop (
    input        clk,
    input        rst,
    input        mode,      // 0: right, 1: left
    output reg [7:0] cnt
);

always @(posedge clk) begin
    if (rst) begin
        cnt <= 8'b0000_0001;
    end
    else if (mode) begin
        // 왼쪽 회전
        if ((cnt == 8'h00) || (cnt == 8'h80))
            cnt <= 8'b0000_0001;
        else
            cnt <= (cnt << 1) | {7'b0, cnt[7]};
    end
    else begin
        // 오른쪽 회전
        if ((cnt == 8'h00) || (cnt == 8'h01))
            cnt <= 8'b1000_0000;
        else
            cnt <= (cnt >> 1) | {cnt[0], 7'b0};
    end
end

endmodule
코드 — for 방식
module LRS_ring_cnt8b_for (
    input        clk,
    input        rst,
    input        mode,      // 0: right, 1: left
    output reg [7:0] cnt
);

integer i;

always @(posedge clk) begin
    if (rst) begin
        cnt <= 8'b0000_0001;
    end
    else if (mode) begin
        if ((cnt == 8'h00) || (cnt == 8'h80)) begin
            cnt <= 8'b0000_0001;
        end
        else begin
            for (i = 7; i > 0; i = i - 1)
                cnt[i] <= cnt[i-1];
            cnt[0] <= cnt[7];
        end
    end
    else begin
        if ((cnt == 8'h00) || (cnt == 8'h01)) begin
            cnt <= 8'b1000_0000;
        end
        else begin
            for (i = 0; i < 7; i = i + 1)
                cnt[i] <= cnt[i+1];
            cnt[7] <= cnt[0];
        end
    end
end

endmodule
코드 — Concatenation 방식
module LRS_ring_cnt8b_conop (
    input        clk,
    input        rst,
    input        mode,      // 0: right, 1: left
    output reg [7:0] cnt
);

always @(posedge clk) begin
    if (rst) begin
        cnt <= 8'b0000_0001;
    end
    else if (mode) begin
        if ((cnt == 8'h00) || (cnt == 8'h80))
            cnt <= 8'b0000_0001;
        else
            cnt <= {cnt[6:0], cnt[7]};
    end
    else begin
        if ((cnt == 8'h00) || (cnt == 8'h01))
            cnt <= 8'b1000_0000;
        else
            cnt <= {cnt[0], cnt[7:1]};
    end
end

endmodule

12.1.5 Parallel Load + 좌우 Shift

load=1이면 8비트 din이 한 번에 register에 저장되고, load=0이면 mode에 따라 좌/우 logical shift가 수행된다.

코드 — 병렬로드/좌우 시프트
module shift_reg8b_pld (
    input        clk,
    input        rst,
    input        load,
    input        mode,      // 0: left, 1: right
    input  [7:0] din,
    output [7:0] pout
);

reg [7:0] reg_tmp;
assign pout = reg_tmp;

always @(posedge clk) begin
    if (rst)
        reg_tmp <= 8'h00;
    else if (load)
        reg_tmp <= din;
    else if (!mode)
        reg_tmp <= reg_tmp << 1;
    else
        reg_tmp <= reg_tmp >> 1;
end

endmodule

12.1.6 8비트 LFSR 의사난수 발생기

12.1의 마지막 실습은 CHAPTER 11의 8비트 LFSR 그림을 참조하지만 현재 장에는 tap 코드가 직접 제공되지 않는다. 따라서 아래 코드는 빈칸의 문자 그대로의 복원본이 아니라 검증용 구현이다.

All-zero lock
XOR feedback LFSR은 8'h00으로 초기화하면 영원히 0에서 빠져나오지 못한다. 따라서 non-zero seed를 사용한다.
8비트 LFSR — 최대길이 계열 예시
LFSR tap은 x^8 + x^6 + x^5 + x^4 + 1 기준으로 사용한다.
module lfsr8 (
    input        clk,
    input        rst,
    output reg [7:0] lfsr
);

// 교육용 보완 구현:
// x^8 + x^6 + x^5 + x^4 + 1 형태의 feedback 예
wire feedback = lfsr[7] ^ lfsr[5] ^ lfsr[4] ^ lfsr[3];

always @(posedge clk) begin
    if (rst)
        lfsr <= 8'h01;   // XOR LFSR은 all-zero seed 금지
    else
        lfsr <= {lfsr[6:0], feedback};
end

endmodule

12.2 계수기 설계

12.2.1 8비트 증가 Counter

현재 cnt가 Adder의 입력으로 들어가고 +1 결과가 다음 clock edge에서 동일한 register에 다시 저장되는 전형적인 순차회로이다.

현재 cnt→+1 Adder→ posedge clkcnt Register
Bit Width
8비트 unsigned counter는 0~255를 표현한다. 255에서 1을 더하면 상위 carry가 잘려 0으로 돌아간다.
코드 — 8비트 증가 Counter
module counter_up (
    input        clk,
    input        rst,
    output reg [7:0] cnt
);

always @(posedge clk) begin
    if (rst)
        cnt <= 8'h00;
    else
        cnt <= cnt + 8'd1;
end

endmodule

12.2.2 8비트 Up/Down Counter

mode=0이면 증가, mode=1이면 감소한다. 하드웨어는 +1/-1 경로와 mode MUX, 8비트 register로 이해할 수 있다.

코드 — 증가/감소 Counter
module ud_cnt8b (
    input        clk,
    input        rst,
    input        mode,      // 0: up, 1: down
    output reg [7:0] cnt
);

always @(posedge clk) begin
    if (rst)
        cnt <= 8'h00;
    else if (!mode)
        cnt <= cnt + 8'd1;
    else
        cnt <= cnt - 8'd1;
end

endmodule

12.2.3 BCD Counter와 FND 표시

BCD counter는 4비트이지만 전체를 쓰지 않고 0~9만 사용한다. cnt==9에서 다음 값을 0으로 만들어 10진수 한 자리 counter가 된다.

코드 — BCD Counter
module bcd_counter (
    input        clk,
    input        rst,
    output reg [3:0] cnt
);

always @(posedge clk) begin
    if (rst)
        cnt <= 4'd0;
    else if (cnt == 4'd9)
        cnt <= 4'd0;
    else
        cnt <= cnt + 4'd1;
end

endmodule
공통 함수 — BCD→FND
BCD→FND 변환 함수는 active-low 8비트 segment table을 모듈 내부에 포함한다.
function [7:0] bcd_to_fnd;
input [3:0] bcd;
begin
    // [7:0] = {DP, G, F, E, D, C, B, A}, active-low 예
    case (bcd)
        4'd0: bcd_to_fnd = 8'b1100_0000;
        4'd1: bcd_to_fnd = 8'b1111_1001;
        4'd2: bcd_to_fnd = 8'b1010_0100;
        4'd3: bcd_to_fnd = 8'b1011_0000;
        4'd4: bcd_to_fnd = 8'b1001_1001;
        4'd5: bcd_to_fnd = 8'b1001_0010;
        4'd6: bcd_to_fnd = 8'b1000_0010;
        4'd7: bcd_to_fnd = 8'b1111_1000;
        4'd8: bcd_to_fnd = 8'b1000_0000;
        4'd9: bcd_to_fnd = 8'b1001_0000;
        default: bcd_to_fnd = 8'b1111_1111;
    endcase
end
endfunction
코드 — BCD Counter + FND Top
자료는 별도 함수 파일 include 방식이지만, 전체 코드 확인이 쉽도록 함수까지 모듈 내부에 넣은 단일 파일 형태다.
module bcd_cnt_fnd (

    input        rst,
    output [7:0] fnd_scan,

);

input [3:0] bcd;
begin
    case (bcd)
        4'd0: bcd_to_fnd = 8'b1100_0000;
        4'd1: bcd_to_fnd = 8'b1111_1001;
        4'd2: bcd_to_fnd = 8'b1010_0100;
        4'd3: bcd_to_fnd = 8'b1011_0000;
        4'd4: bcd_to_fnd = 8'b1001_1001;
        4'd5: bcd_to_fnd = 8'b1001_0010;
        4'd6: bcd_to_fnd = 8'b1000_0010;
        4'd7: bcd_to_fnd = 8'b1111_1000;
        4'd8: bcd_to_fnd = 8'b1000_0000;
        4'd9: bcd_to_fnd = 8'b1001_0000;
        default: bcd_to_fnd = 8'b1111_1111;
    endcase

bcd_counter u_counter (
    .clk(clk),
    .rst(rst),
    .cnt(cnt)
);

assign fnd_data = bcd_to_fnd(cnt);

// 한 자리만 활성화하는 active-low scan 예
assign fnd_scan = 8'b1111_1110;

FND 극성과 비트 순서
위 패턴은 {DP,G,F,E,D,C,B,A} 순서의 active-low segment를 가정한 기준 코드이다. 실제 실습장비의 공통 애노드/공통 캐소드 및 bit 순서가 다르면 segment table만 조정한다.

12.3 주파수 분주기 설계

12.3.1 1/10 Frequency Divider

10Hz 입력에서 1Hz, 50% duty cycle을 만들려면 출력 상태를 입력 clock 5주기마다 반전하면 된다. 따라서 전체 출력 한 주기는 입력 clock 10주기가 된다.

mclk 5회→clk_1hz toggle+ mclk 5회다시 toggle
코드 — 1/10 분주기
분주 출력 자체는 외부 관찰용 신호이므로 register로 생성한다. reset은 mclk에 동기화했다.
module frq_div10 (
    input        mclk,
    input        rst,
    output       mclk_out,
    output reg   clk_1hz
);

reg [2:0] cnt;

assign mclk_out = mclk;

always @(posedge mclk) begin
    if (rst) begin
        cnt     <= 3'd0;
        clk_1hz <= 1'b0;
    end
    else if (cnt == 3'd4) begin
        cnt     <= 3'd0;
        clk_1hz <= ~clk_1hz;
    end
    else begin
        cnt <= cnt + 3'd1;
    end
end

endmodule
코드 — 테스트벤치
`timescale 1ns/1ps

module tb_frq_div10;

reg rst;
reg mclk;
wire mclk_out;
wire clk_1hz;

frq_div10 dut (
    .mclk    (mclk),
    .rst     (rst),
    .mclk_out(mclk_out),
    .clk_1hz (clk_1hz)
);

initial begin
    mclk = 1'b0;
    forever #10 mclk = ~mclk;
end

initial begin
    rst = 1'b1;
    repeat (2) @(posedge mclk);
    rst = 1'b0;

    repeat (40) @(posedge mclk);
    $finish;
end

endmodule

12.3.2 여덟 가지 동작모드 LED Shifter

LED Shifter는 내부 분주 clock을 별도 clock으로 사용하지 않고 clock-enable 방식으로 구성한다.

Generated Clock 주의

    input        mclk,       // 100 Hz
    input        rst,
    input  [3:0] mode,
    output reg [7:0] led_out
);

reg [7:0] div_cnt;
reg [7:0] terminal_count;

// mode[2:0] : 000=100Hz, 001=10Hz, 010=1Hz, 100=0.5Hz
always @(*) begin
    case (mode[2:0])
        3'b000: terminal_count = 8'd0;
        3'b001: terminal_count = 8'd9;
        3'b010: terminal_count = 8'd99;
        3'b100: terminal_count = 8'd199;

    endcase
end

always @(posedge mclk) begin
    if (rst) begin
        div_cnt  <= 8'd0;
        led_out  <= 8'b0000_0001;
    end
    else begin
        if (div_cnt >= terminal_count) begin
            div_cnt <= 8'd0;

            end
            else begin
                led_out <= {led_out[0], led_out[7:1]};
            end
        end
        else begin
            div_cnt <= div_cnt + 8'd1;
        end
    end

module keypad_scan (
    input        clk,
    input        rst,
    input  [3:0] key_row,
    output [2:0] key_col,
    output reg [11:0] key_data
);

localparam NO_SCAN = 3'b000;
localparam COL1    = 3'b001;
localparam COL2    = 3'b010;
localparam COL3    = 3'b100;

reg [2:0] state;
reg [3:0] row_meta;
reg [3:0] row_sync;

wire key_stop = |row_sync;
assign key_col = state;

// 외부 키패드 입력을 clk 도메인으로 2-FF 동기화
always @(posedge clk) begin
    if (rst) begin
        row_meta <= 4'b0000;
        row_sync <= 4'b0000;
    end
    else begin
        row_meta <= key_row;
        row_sync <= row_meta;
    end
end

// Column scan FSM
always @(posedge clk) begin
    if (rst) begin
        state <= NO_SCAN;
    end
    else if (!key_stop) begin
        case (state)
            NO_SCAN: state <= COL1;
            COL1   : state <= COL2;
            COL2   : state <= COL3;
            COL3   : state <= COL1;
            default: state <= NO_SCAN;
        endcase
    end
end

// key_data 생성
always @(posedge clk) begin
    if (rst) begin
        key_data <= 12'b0;
    end
    else begin
        case (state)
            COL1: begin
                case (row_sync)
                    4'b0001: key_data <= 12'b0000_0000_0001; // 1
                    4'b0010: key_data <= 12'b0000_0000_1000; // 4
                    4'b0100: key_data <= 12'b0000_0100_0000; // 7
                    4'b1000: key_data <= 12'b0010_0000_0000; // *
                    default: key_data <= 12'b0;
                endcase
            end

            COL2: begin
                case (row_sync)
                    4'b0001: key_data <= 12'b0000_0000_0010; // 2
                    4'b0010: key_data <= 12'b0000_0001_0000; // 5
                    4'b0100: key_data <= 12'b0000_1000_0000; // 8
                    4'b1000: key_data <= 12'b0100_0000_0000; // 0
                    default: key_data <= 12'b0;
                endcase
            end

            COL3: begin
                case (row_sync)
                    4'b0001: key_data <= 12'b0000_0000_0100; // 3
                    4'b0010: key_data <= 12'b0000_0010_0000; // 6
                    4'b0100: key_data <= 12'b0001_0000_0000; // 9
                    4'b1000: key_data <= 12'b1000_0000_0000; // #
                    default: key_data <= 12'b0;
                endcase
            end

            default:
                key_data <= 12'b0;
        endcase
    end
end

endmodule

12.5.1 Beep 발생기

1kHz 입력 clock을 Piezo 구동 주파수로 사용하고, 0.1초 단위의 on/off envelope를 만들어 삐 소리와 묵음을 반복한다. 1kHz system clock 하나만 사용하고 100ms tick을 clock-enable로 만든다.

1kHz clk→ 100회 계수100ms tick→pwm_cnt →beep_on
코드 — Piezo Beep
BEEP_PERIOD=5이면 0.5초 tone + 0.5초 mute가 반복된다.
module piezo_beep #(
    parameter integer BEEP_PERIOD = 5   // 1~9
) (
    input  clk,     // 1 kHz
    input  rst,
    output piezo_out
);

reg [6:0] tick_cnt;   // 0~99
reg [3:0] pwm_cnt;    // 0~9

wire beep_on = (pwm_cnt < BEEP_PERIOD);

always @(posedge clk) begin
    if (rst) begin
        tick_cnt <= 7'd0;
        pwm_cnt  <= 4'd0;
    end
    else if (tick_cnt == 7'd99) begin
        tick_cnt <= 7'd0;

        if (pwm_cnt == 4'd9)
            pwm_cnt <= 4'd0;
        else
            pwm_cnt <= pwm_cnt + 4'd1;
    end
    else begin
        tick_cnt <= tick_cnt + 7'd1;
    end
end

// 내부 클록으로 사용하지 않고 외부 Piezo 구동 신호로만 gating
assign piezo_out = beep_on & clk;

endmodule

    input        rst,
    input  [7:0] key_in,
    output reg   piezo_freq
);

localparam [9:0] D4_TONE = 10'd851; // 약 293 Hz
localparam [9:0] E4_TONE = 10'd758; // 약 329 Hz
localparam [9:0] F4_TONE = 10'd716; // 약 349 Hz
localparam [9:0] G4_TONE = 10'd638; // 약 392 Hz
localparam [9:0] A4_TONE = 10'd568; // 약 440 Hz
localparam [9:0] B4_TONE = 10'd506; // 약 494 Hz
localparam [9:0] C5_TONE = 10'd477; // 약 523 Hz

reg [9:0] limit;
reg       tone_en;
reg [9:0] cnt;

always @(*) begin
    tone_en = 1'b1;

    case (key_in)
        8'b0000_0001: limit = C4_TONE; // key 1
        8'b0000_0010: limit = D4_TONE; // key 2
        8'b0000_0100: limit = E4_TONE; // key 3
        8'b0000_1000: limit = F4_TONE; // key 4
        8'b0001_0000: limit = G4_TONE; // key 5
        8'b0010_0000: limit = A4_TONE; // key 6
        8'b0100_0000: limit = B4_TONE; // key 7
        8'b1000_0000: limit = C5_TONE; // key 8
        default: begin
            limit   = 10'd0;
            tone_en = 1'b0;
        end
    endcase
end

    if (rst) begin
        cnt        <= 10'd0;
        piezo_freq <= 1'b0;
    end

        cnt <= cnt + 10'd1;
    end

12.6 스텝 모터 구동회로 설계

12.6.1 스텝 모터 구동 구조

자료는 1kHz→50Hz 분주 clock을 만들고, magnetic sensor 출력을 IBUFG에 넣어 별도 clock으로 사용한다. 교육적으로 동작을 이해하기에는 쉽지만 실제 RTL에서는 sensor를 clock으로 사용하지 않고 system clock으로 동기화한 뒤 edge detect하는 편이 안정적이다.

CDC/Clocking 문제
mag_sen은 외부 sensor이므로 FPGA clock과 비동기이다. 이를 그대로 always @(posedge mag_sen)의 clock으로 사용하면 새로운 clock domain이 만들어지고 timing constraint와 metastability 관리가 어려워진다.

12.6.2 2상 여자 Sequence

motor_data 순서를 정의 {A, B, /A, /B}로 두면 2상 여자 4-step 예시는 다음과 같이 표현할 수 있다.

Step{A,B,/A,/B}반대 방향
011001100
101101001
200110011
310010110
공통 Core — 단일 clock + sensor synchronizer
1kHz system clock에서 20 clock마다 step_ce를 발생시켜 50 step/s로 동작하는 공통 Core다.
module motor_drive_core #(
    parameter integer TURN_LIMIT = 5
) (
    input        clk,         // 1 kHz
    input        rst,
    input        mag_sen,
    output reg [3:0] motor_data,
    output reg [3:0] turn_count,
    output reg       direction
);

reg [4:0] step_div;
reg [1:0] step_idx;

// 비동기 외부 센서 동기화 + 상승에지 검출
reg mag_meta, mag_sync, mag_d;

wire mag_rise = mag_sync & ~mag_d;
wire step_ce  = (step_div == 5'd19);  // 1kHz / 20 = 50 step/s

always @(posedge clk) begin
    if (rst) begin
        mag_meta <= 1'b0;
        mag_sync <= 1'b0;
        mag_d    <= 1'b0;
    end
    else begin
        mag_meta <= mag_sen;
        mag_sync <= mag_meta;
        mag_d    <= mag_sync;
    end
end

always @(posedge clk) begin
    if (rst)
        step_div <= 5'd0;
    else if (step_ce)
        step_div <= 5'd0;
    else
        step_div <= step_div + 5'd1;
end

always @(posedge clk) begin
    if (rst) begin
        step_idx   <= 2'd0;
        motor_data <= 4'b1100;
    end
    else if (step_ce) begin
        // motor_data = {A, B, /A, /B}
        if (!direction) begin
            case (step_idx)
                2'd0: motor_data <= 4'b1100;
                2'd1: motor_data <= 4'b0110;
                2'd2: motor_data <= 4'b0011;
                2'd3: motor_data <= 4'b1001;
            endcase
        end
        else begin
            case (step_idx)
                2'd0: motor_data <= 4'b1100;
                2'd1: motor_data <= 4'b1001;
                2'd2: motor_data <= 4'b0011;
                2'd3: motor_data <= 4'b0110;
            endcase
        end

        step_idx <= step_idx + 2'd1;
    end
end

always @(posedge clk) begin
    if (rst) begin
        turn_count <= 4'd0;
        direction  <= 1'b0;
    end
    else if (mag_rise) begin
        if (turn_count == TURN_LIMIT - 1) begin
            turn_count <= 4'd0;
            direction  <= ~direction;
        end
        else begin
            turn_count <= turn_count + 4'd1;
        end
    end
end

endmodule

                    line_idx <= 0;
                    state <= S_LINE3;
                end
                else cnt_2s <= cnt_2s + 1'b1;
            end

            S_LINE3: begin
                if (line_idx == 16) begin
                    line_idx <= 0;
                    state <= S_LINE4;
                end
                else line_idx <= line_idx + 1'b1;
            end

            S_LINE4: begin
                if (line_idx == 16) begin
                    line_idx <= 0;
                    cnt_2s <= 0;
                    state <= S_DELAY_2S_B;
                end
                else line_idx <= line_idx + 1'b1;
            end

            S_DELAY_2S_B: begin
                if (cnt_2s == 399) begin
                    cnt_2s <= 0;
                    state <= S_CLEAR;
                end
                else cnt_2s <= cnt_2s + 1'b1;
            end

            S_CLEAR: begin
                line_idx <= 0;
                state <= S_LINE1;
            end

            default: state <= S_DELAY_100MS;
        endcase
    end
end

always @(*) begin
    lcd_rs   = 1'b0;
    lcd_data = 8'h00;
    lcd_e    = 1'b0;

    case (state)
        S_FUNCTION: begin
            lcd_data = 8'h38;
            lcd_e = (cnt_5ms >= 1 && cnt_5ms <= 3);
        end
        S_ENTRY: begin
            lcd_data = 8'h06;
            lcd_e = (cnt_5ms >= 1 && cnt_5ms <= 3);
        end
        S_DISP_ON: begin
            lcd_data = 8'h0C;
            lcd_e = (cnt_5ms >= 1 && cnt_5ms <= 3);
        end

        S_LINE1, S_LINE2, S_LINE3, S_LINE4: begin
            if (line_idx == 0) begin
                lcd_rs = 1'b0;
                if ((state == S_LINE1) || (state == S_LINE3))
                    lcd_data = ADDRESS_LINE1;
                else
                    lcd_data = ADDRESS_LINE2;
            end
            else begin
                lcd_rs = 1'b1;
                case (state)
                    S_LINE1: lcd_data = data_line1(line_idx - 1'b1);
                    S_LINE2: lcd_data = data_line2(line_idx - 1'b1);
                    S_LINE3: lcd_data = data_line3(line_idx - 1'b1);
                    S_LINE4: lcd_data = data_line4(line_idx - 1'b1);
                    default: lcd_data = 8'h20;
                endcase
            end
            lcd_e = (cnt_5ms >= 1 && cnt_5ms <= 3);
        end

        S_CLEAR: begin
            lcd_data = 8'h01;
            lcd_e = (cnt_5ms >= 1 && cnt_5ms <= 3);
        end

        default: begin
            lcd_rs = 1'b0;
            lcd_data = 8'h00;
            lcd_e = 1'b0;
        end
    endcase
end

endmodule

LCD Enable Timing
1kHz clock의 5개 cycle을 한 전송 slot(5ms)으로 보고 cnt_5ms=1~3 구간에서 lcd_e를 HIGH로 만든다. 주소 또는 문자 데이터는 해당 slot 동안 안정적으로 유지된다.

이 장핵심 설계 관점
순차회로는 항상 현재 Register 값 → 조합논리 → 다음 값 → posedge clk에서 저장 흐름으로 읽는다. 특히 FPGA 구현에서는 clock 자체를 logic처럼 여기지 말고, 가능한 한 보드의 주 clock 하나를 유지하면서 clock-enable을 이용해 느린 동작을 만드는 습관이 중요하다.
